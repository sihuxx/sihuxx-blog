#typescript 

### 01 Array (배열)

#### 예제 (1) 타입 제한

```typescript
let numbers: number[] = [1, 2, 3, 4, '5'];
// numbers의 타입을 number로 지정하여 배열 요소들의 타입을 지정함
// 위 처럼 마지막 요소를 문자열로 바꾸면 에러가 발생함

let fruits: Array<string> = ["Apple", "Banana", "Orange"];
// 배울 제네릭을 사용한 타입 제한 -> 대괄호 사용 방식과 기능은 동일

let scores: number[] = [];
// 타입만 지정하고 요소는 후에 추가하게
scores.push(95); // 숫자 요소 추가 (O)
scores.push(88); // 숫자 요소 추가 (O)

scores.push("A+"); // 문자 요소 추가 (X)
// scores 배열의 타입이 숫자기 때문에 문자열이 들어올 수 없다

let colors = ["Red", "Green", "Blue"];
colors.push(123);
// colors 배열의 타입이 문자열이기 때문에 숫자가 들어갈 수 없다

console.log(numbers[0], fruits[1], colors[2]); // 1 Banana Blue
```

#### 예제 (2) 다차원 배열

```typescript
// 이차원 배열
let matrix: number[][] = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]

// 삼차원 배열

    [ [5, 6], [7, 8] ]

// 안의 내부 요소를 숫자열이 아닌 다른 타입의 요소로 변경하면 오류가 발생한다
```

#### 예제 (3) 배열 메소드
```typescript
const names: string[] = ['Alice', 'Bob', 'Charlie', 'David'];
// 문자열 배열이기 때문에 문자열 메소드만 사용이 가능

const nameLengths: number[] = names.map(name => name.length);
// 배열의 요소들을 각 요소의 글자 길이로 변환한 후 새 배열에 반환
console.log(nameLengths);

const longName: string[] = names.filter(name => name.length > 4);
// 배열의 요소둘 중 글자 길이가 4보다 큰 요소만 골라내서 새 배열에 반환
console.log(longName);

const foundName: string | undefined = names.find(name => name.startsWith('B'));
// 'B'로 시작하는 첫 문자열을 찾아 반환
console.log(foundName);
```

#### 예제 (4) 읽기 전용 배열
```typescript
const originalArray: number[] = [1, 2, 3, 4, 5];
originalArray[0] = 10;

const readOnlyNumbers: ReadonlyArray<number> = originalArray; // 제네릭 타입 읽기 전용 배열

const readOnlyScores: readonly number[] = [90, 85, 95]; // readonly 연산자 읽기 전용 배열

readOnlyNumbers[0] = 20; // 요소 수정 => 오류
readOnlyScores.push(100); // 요소 추가 => 오류

const newArray = [...readOnlyNumbers, 6] // [1, 2, 3, 4, 5, 6]
// 읽기 전용 배열을 새 배열로 복사하는 건 가능

newArray[0] = 0; 
newArray.push(7);
// 읽기 전용 배열이 아니기에 요소 수정이나 추가가 가능

console.log(newArray);
```

### 02 Tuple (튜플)
**지정석이 있는 열차칸**
배열처럼 생겼지만 각 요소의 타입과 순서가 정해져 있고 이를 어기면 컴파일 오류 발생

#### 예제 (1) 타입 지정
```typescript
let person : [string, number] = ['John', 30];
// John이라는 문자열과 30이란 숫자를 person이라는 변수에 담는 코드
// 배열 타입에서 각 자리에 들어올 타입을 string과 number로 제한
console.log(person[0]); // John
console.log(person[1]); // 30

person = [30, "John"]; 
// 튜플에 지정된 타입과 다른 타입의 값을 넣거나, 지정한 개수를 초과하는 값이 들어오면 오류 발생

const [firstName, age] = person; // 구조 분해 할당 가능
// firstName에 person 배열의 첫 번째 값인 "John", age에 person 배열의 두 번째 값인 30을 넣음
console.log(firstName); // John
console.log(age); // 30

type OptionalTuple = [string, number, boolean?];
// boolean에 물음표를 붙암으로서 해당 요소를 넣어도 되고, 안 넣어도 되는 요소로 지정
const complete: OptionalTuple = ['Jane', 25, true];
const partial: OptionalTuple = ["Mike", 40];
```

#### 예제 (2) 함수 반환값
```typescript
function getUserInfo(): [string, number, boolean] {
    return ["alex_dev", 28, false];
    // 문자열, 숫자, 불리언
}

const [userName, userAge, isAdmin] = getUserInfo(); // 구조 분해 할당
// 순서대로 "alex_dev", 28, false 들어감

console.log(
    `Username: ${userName}, Age: ${userAge}, Admin: ${isAdmin}`
    // `Username: alex_dev, Age: 28, Admin: false`
);
```


#### 예제 (3) 타입 추론
```typescript
const inferredArray = [1, 'hello'];
// inferredArray: (string | number)[] 타입 추론

const inferredTuple = [1, 'hello'] as const;
// '1'이라는 리터럴 타입, 'hello'라는 리터럴 타입의 튜플이라 추론
// => 이 값 외에는 어느 다른 값도 들어올 수 없다

inferredTuple[0] = 2
inferredTuple[1] = 'world'
// 같은 타입이라도 해당 자리에 다른 값을 넣으려고 하면 컴파일 오류가 발생
// => 값 자체가 타입이 됨

let explicitTuple: [number, string] = [1, 'hello'];
explicitTuple = [2, 'world']
// 일반적인 튜플, 각 자리에 오는 요소들의 타입이 조건 타입과 일치하면 담을 수 있음
```


### 03 Object (객체)

#### 예제 (1) 기본 객체
```typescript
const person1: { name: string; age: number } = {
    // name과 age의 타입이 각각 string, number임을 명시
    name: "John",
    age: 25
}

const wrongPerson1: { name: string; age: number } = {
    name: "John",
    age: "25" // 명시한 타입과 다른 타입 => 오류
}

const wrongPerson2: { name: string; age: number } = {
    name: "Mark", // 객체 타입 속성 중 age 요소 누락 => 오류
    married: true // 명시한 요소 중 없는 요소 추가 => 오류
}

person1.name = "Jane";
person1.age = "30"; // 명시한 타입과 다른 타입으로 수정 => 오류

// 명시한 요소의 타입과 다른 타입을 가지거나, 그 요소를 누락하거나, 없는 요소를 추가하면 오류 발생
```


#### 예제 (2) 인터페이스
```typescript
interface User { // User란 이름의 인터페이스 정의
    id: number, // 숫자열 타입 명시
    username: string, // 문자열 타입 명시
    isAdmin: boolean, // 불리언 타입 명시
}
// 한 번 인터페이스를 정의해두면 같은 구성의 객체를 만들 때 매 번 각 속성과 그 타입을 정의하지 않아도 됨

const adminUser: User = { // 옳은 예시
    id: 0,
    username: "admin",
    isAdmin: true,
}

const user1: User1 = { // 정의된 바와 다르게 속성값 추가
    id: 1,
    username: "admin",
    // isAdmin 속성 누락
}

const user2: User = { 
    id: "2", // id 속성에 문자열 넣음
    username: "admin",
    isAdmin: false,
}
```
