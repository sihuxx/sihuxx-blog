## JS

```js
const mottos = [...document.querySelectorAll(".mottos > div")]
const dess = e => document.querySelector(`.desContianer > .${e} `)

mottos.forEach((e) => {
  e.addEventListener("mouseenter", () => {
    $$('.mottoName').forEach(t => {
      t.style.opacity = 0
    })
    dess(e.className).style.opacity = 1
    mottos.forEach((e2) => e2.style.backgroundImage = `url("../asset/motto/${e.className}.jpg")`)
    e.querySelector('.mottoName').style.opacity = 1
  })

  e.addEventListener("mouseleave", () => {
    $$('.mottoName').forEach(t => {
      t.style.opacity = 1
    })
    dess(e.className).style.opacity = 0
    mottos.forEach((e2) => e2.style.backgroundImage = `url("../asset/motto/${e2.className}.jpg")`)
  })
})
```
