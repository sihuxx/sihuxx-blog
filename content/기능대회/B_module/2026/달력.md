# 달력 및 스케줄 관리 시스템

## 1. HTML 구조 (index.html)

```html

<select id="month">
  <option value="1">1월</option><option value="2">2월</option>

  <thead>
    <tr>
      <th>일</th><th>월</th><th>화</th><th>수</th><th>목</th><th>금</th><th>토</th>
    </tr>

```

## 2. JavaScript 로직 (script.js)

파일을 읽어와 **2026년 4월에만** 스케줄 데이터를 표시하는 핵심 로직

```javascript
const $ = (e) => document.querySelector(e);
const [yS, mS, body] = [$('#year'), $('#month'), $('#calendar-body')];

/** 1. 스케줄 데이터 파싱 **/
// fetch로 텍스트 파일을 가져옴
const resp = await fetch("./B-Module/schedule.txt").then(res => res.text());
const schedule = {};

// 빈 줄(\n\s*\n)을 기준으로 데이터 블록 분리
resp.split(/\n\s*\n/).forEach(block => {
  const lines = block.split('\n').map(l => l.trim());
  const day = parseInt(lines[0]); // 첫 줄은 날짜(숫자)
  schedule[day] = lines.slice(1); // 나머지는 이벤트 배열
});

/** 2. 달력 렌더링 함수 **/
function render() {
  const [y, m] = [+yS.value, +mS.value]; // 선택된 년, 월 (숫자로 변환)
  const startDay = new Date(y, m - 1, 1).getDay(); // 해당 월 시작 요일 (0:일 ~ 6:토)
  const lastDate = new Date(y, m, 0).getDate();    // 해당 월의 마지막 날짜

  let html = '<tr>' + '<td></td>'.repeat(startDay); // 1일 앞의 빈칸 생성

  for (let d = 1; d <= lastDate; d++) {
    // 토요일(인덱스 6)이 지나면 줄바꿈
    if ((d + startDay - 1) % 7 === 0 && d !== 1) html += '</tr><tr>';

    // 조건: 2026년 4월인 경우에만 스케줄 데이터 추출
    let content = "";
    if (y === 2026 && m === 4 && schedule[d]) {
      content = schedule[d].map(msg => `<div class="event">${msg}</div>`).join('');
    }

    html += `<td><div class="date-num">${d}</div>${content}</td>`;
  }

  body.innerHTML = html + '</tr>';
}

/** 3. 이벤트 바인딩 및 초기화 **/
// 값이 바뀔 때마다 다시 그리기
[yS, mS].forEach(select => select.onchange = render);

// 초기값: 2026년 4월
yS.value = 2026;
mS.value = 4;
render();
```
