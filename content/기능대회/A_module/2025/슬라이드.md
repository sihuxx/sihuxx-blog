## JS

```js
const slideList = document.querySelector('.slide ul');
const slideItems = document.querySelectorAll('.slide li');

let currentIndex = 0;
const slideCount = slideItems.length;

function moveSlide() {
  currentIndex++;

  if (currentIndex >= slideCount) {
    currentIndex = 0;
  }

  slideList.style.transform = `translateX(-${currentIndex * 25}%)`;
}

setInterval(moveSlide, 3000);
```
